## Implementation Details

### Card Component
```tsx
export function Card({ children, variant = 'elevated', style }) {
  const shadows = {
    elevated: { boxShadow: '0 2px 8px rgba(0, 0, 0, 0.06)' },
    outlined: { borderWidth: 1, borderColor: colors.surface[200] },
    flat: {},
  };
  
  return (
    <Animated.View style={[
      {
        backgroundColor: '#FFFFFF',
        borderRadius: radii.lg,
        borderCurve: 'continuous',
        padding: spacing.md,
        ...shadows[variant],
      },
      style
    ]}>
      {children}
    </Animated.View>
  );
}
```
